//enter password from the user maximum 20 digits/character press enter key if you exit?
#include 
int main() {
    char pass[21];
    int i = 0;
    char ch;
    printf("Enter password (max 20 chars, press Enter to exit): ");
    while (i < 20) {
        ch = getchar();     // works in VS Code
        if (ch == '\n')     // Enter key
            break;
        pass[i] = ch;
        i++;
    }
    pass[i] = '\0';
    printf("\nYour password is: %s\n", pass);
    return 0;
}   /*o/p:
    Enter password (max 20 chars, press Enter to exit): mySecret123
    Your password is: mySecret123 */
