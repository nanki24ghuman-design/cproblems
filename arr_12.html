//find maximum values of array?
#include<stdio.h>
void main()
{
int arr[5];
int i,max;
printf("Enter 5 numbers  ");
for(i=0;i<5;i++){
scanf("%d",&arr[i]);
}
max=arr[0];
for(i=1;i<5;i++){
if(arr[i]>max){
max=arr[i];
}}
printf("maximum value of array is %d",max);
return 0;
}/*Enter 5 elements: 4 7 1 9 3  
Maximum value: 9  */
