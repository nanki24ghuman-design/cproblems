//2. Enter the matrix from and transpose them ?
#include 
int  main() {
    int a[2][2], t[2][2], i, j;
    printf("Enter 4 numbers for matrix:\n");
    for(i=0; i<2; i++) //row
        for(j=0; j<2; j++) //column
            scanf("%d", &a[i][j]);

        for(i=0; i<2; i++)
        for(j=0; j<2; j++)
            t[j][i] = a[i][j];

    printf("\nTranspose of Matrix:\n");
    for(i=0; i<2; i++) {
        for(j=0; j<2; j++)
            printf("%d ", t[i][j]);
        printf("\n");
    }
   return 0;
}
