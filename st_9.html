//9. Enter a string from user and copy that string into another string ?
#include <stdio.h>
#include <string.h>
void main() {
    char str1[100], str2[100];
    printf("Enter a string: ");
    fgets(str1,100,stdin);

    strcpy(str2, str1);  // copy str1 into str2

    printf("\nCopied string: %s", str2);
    return 0;
}
