//1. Enter a string from user and convert in uppercase ?
#include <stdio.h>
#include <string.h>

void main() {
    char str[100];
    int i;
    printf("Enter a string: ");
    fgets(str ,100,stdin);

    for(i=0; str[i]!='\0'; i++) {
        if(str[i]>='a' && str[i]<='z')
            str[i] = str[i] - 32;
    }

    printf("Uppercase string: %s", str);
    return 0;
}
