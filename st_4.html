// 4.Enter a string from user and convert into title case ?
#include <stdio.h>
void main() {
    char str[100];
    int i;
    printf("Enter a string: ");
    scanf("%s",str);

    // convert first letter to uppercase if needed
    if(str[0]>='a' && str[0]<='z')
        str[0] = str[0] - 32;

    // convert letters after space to uppercase
    for(i=1; str[i]!='\0'; i++) {
        if(str[i-1]==' ' && str[i]>='a' && str[i]<='z')
            str[i] = str[i] - 32;
        else if(str[i-1]!=' ' && str[i]>='A' && str[i]<='Z')
            str[i] = str[i] + 32;  // make non-first letters lowercase
    }
    printf("Title case: %s", str);
   
    return 0;
}
