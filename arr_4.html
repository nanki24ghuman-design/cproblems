//enter an array from user and copy it into another array in reverse order
#include<stdio.h>
void main()
{
int arr1[20],arr2[20],n,i,j;
printf("enter size of array: ");
scanf("%d",&n);
printf("Enter %d elements: ",n);
for(i=0;i<n;i++){
scanf("%d",&arr1[i]);
}
j=0;
for(i=n-1;i>=0;i--)
{
arr2[j]=arr1[i];
j++;
}
printf("Elements of second array in reversed order are ");
for(i=0;i<n;i++){
printf("%d",arr2[i]);
}
return 0;
}/*Enter size of array: 5  
Enter 5 elements: 1 2 3 4 5  
Reversed copied array: 5 4 3 2 1  */
