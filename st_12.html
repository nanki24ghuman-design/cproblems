//12. Enter the string from user and copy reverse order into another string ?

#include <stdio.h>
#include <string.h>
void main() {
    char str1[100], str2[100];
    int i, j, len;
    printf("Enter a string: ");
    scanf("%s",str1);

    len = strlen(str1);
    j = 0;

    // copy characters from end to start
    for(i = len - 1; i >= 0; i--) {
        str2[j] = str1[i];
        j++;
    }
    str2[j] = '\0';  // null terminate

    printf("\nReversed string: %s", str2);
    return 0;
}
