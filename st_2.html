//2. Enter a string from user and convert into lowercase ?
#include <stdio.h>
#include <string.h>

void main() {
    char str[100];
    int i;
    printf("Enter a string: ");
    scanf("%s",str);

    for(i=0; str[i]!='\0'; i++) {
        if(str[i]>='A' && str[i]<='Z')
            str[i] = str[i] + 32;
    }

    printf("Lowercase string: %s", str);
    return 0;
}
