/*enter student details like roll number,fee,dob and print all the details
in ascending order*/
#include<stdio.h>
struct student{
int roll;
char name[50];
char dob[30];
float fee;
};

void main()
{
struct student s[50],a;
int i,j,n;
float fee;
printf("Enter number of students ");
scanf("%d",&n);
for(i=0;i<n;i++){
printf(" Enter the details of student\n ",i+1);
printf("Enter roll number ");
scanf("%d",&s[i].roll);
printf("Enter the name ");
scanf("%s",s[i].name);
printf("\nEnter the fees ");
scanf("%f",&fee);
printf("enter DOB ");
scanf("%s",s[i].dob);
}
for(i=0;i<n-1;i++)
{   
	for(j=1+1;j<n;j++)
	{
	if(s[i].roll<s[j].roll)
	a=s[i];
	s[i];s[j];
	s[i]=a;
	}
}
printf(" Students details ");
for(i=0;i< n;i++){
printf(" Roll number %d\n",s[i].roll);
printf(" Name %s\n",s[i].name);
printf(" Fees %f\n",s[i].fee);
printf(" DOB %s\n",s[i].dob);
}
return 0;
}/*Enter number of students: 2
