//14. Enter the string from user and enter a character , it is exist or not ?
#include <stdio.h>
#include <string.h>

void main() {
    char str[100], ch;
    int i, found = 0;
    printf("Enter a string: ");
    scanf("%s",str);

    printf("Enter a character to search: ");
    scanf("%c", &ch);

    for(i = 0; str[i] != '\0'; i++) {
        if(str[i] == ch) {
            found = 1;
            break;
        }
    }
    if(found)
        printf("\nCharacter '%c' exists in the string.", ch);
    else
        printf("\nCharacter '%c' does not exist in the string.", ch);
    return 0;
}
