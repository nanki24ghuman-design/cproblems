//enter 5 values of array and sort in ascending order?
#include<stdio.h>
void main()
{
int arr[5];
int i,j,sort;
printf("enter 5 numbers ");
for(i=0;i<5;i++){
scanf("%d",&arr[i]);
}
for(i=0;i<5;i++){
for(j=i+1;j<5;j++){
if(arr[i]>arr[j]){
sort=arr[j];
arr[i]=arr[j];
arr[j]=sort;
}
} }
printf("Array in ascending order ");
for(i=0;i<5;i++){
printf(" %d",arr[i]);
}
return 0;
}/*Enter 5 elements: 50 20 10 30 40  
Sorted array: 10 20 30 40 50  */
