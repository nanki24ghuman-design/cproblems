//17. Enter the string from user and enter bound number and print character on that bound ?
#include <stdio.h>
#include <string.h>

void  main() {
    char str[100];
    int pos, len;
    printf("Enter a string: ");
    scanf("%s",str);

    len = strlen(str);

    printf("Enter position (starting from 0): ");
    scanf("%d", &pos);

    if(pos >= 0 && pos < len)
        printf("\nCharacter at position %d is: %c", pos, str[pos]);
    else
        printf("\nOut of bound!");

    return 0;
}
