// write a program to calculate area and perimeter of rectangle
#include
void main()
{
int l,b,area,perimeter;
printf("Enter the length: ");
scanf("%d",&l);
printf("Enter the breadth:");
scanf("%d",&b);
area=l*b;
printf("Area is %d\n",area);
perimeter=2*(l+b);
printf("Perimeter is %d",perimeter);
}/*Enter the length: 6 
    Enter the breadth:3  
    Area = 18.00  
    Perimeter = 18.00  */
