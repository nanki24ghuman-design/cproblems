//13. Enter two string from user and print difference each character ?

#include <stdio.h>
#include <string.h>
void main() {
    char str1[100], str2[100];
    int i, len1, len2;
    printf("Enter first string: ");
    scanf("%s",str1);

    printf("Enter second string: ");
    scanf("%s",str2);

    len1 = strlen(str1);
    len2 = strlen(str2);

    printf("\nDifference of each character:\n");

    for(i = 0; i < len1 && i < len2; i++)
        printf("%c - %c = %d\n", str1[i], str2[i], str1[i] - str2[i]);

   return 0;
}
