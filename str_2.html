/* create structure of employee which there is emp.id,names and desination
of 50 employees from the user and print only names*/
#include<stdio.h>
struct employee{
int id;
char name[50];
char designation[50];
};
void main()
{
struct employee emp[50];
int i,n;
printf("Enter number of employees ");
scanf("%d",&n);
for(i=0;i<n;i++){
printf("\nEnter details of Employee %d:\n", i + 1);
printf("Employee ID");
scanf("%d",&emp[i].id);
printf("Enter the name ");
scanf("%s",emp[i].name);
printf("\nEnter the designation ");
scanf("%s",emp[i].designation);

}
printf(" Names of Employees  are ");
for(i=0;i<n;i++)
{
printf("%s\n",emp[i].name);
}
return 0;
}
/*Enter number of employees: 2

Enter details of Employee 1:
ID: 101
Name: Raj
Designation: Manager

Enter details of Employee 2:
ID: 102
Name: Ravi
Designation: Clerk
Output:
Raj
Ravi
*/
