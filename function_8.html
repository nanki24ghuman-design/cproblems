//8. Make function for find out Cube a user entered number ?

#include 

void cube(int n) {
    n=n * n * n;
    return n;
}

void main() {
    int res,num;
    printf("Enter a number: ");
    scanf("%d",&num);

    res=cube(num);
    printf("Cube of %d = %d",num,res);
    return 0;
}
/*Enter number: 6  
Cube = 216
Enter number: 8 
Cube = 512 */
