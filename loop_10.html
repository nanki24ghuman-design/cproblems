//10.. Write a program print the table of 4 in the format given as below (till 20)? 4*1=4 and so on.
#include
int main()
{
int i,result=1;
printf("Multiplication table of 4 (till 20):\n");
for(i=1;i<=20;i++)
{
result=4*i;
printf("4*%d=%d\n",i,result);
}
return 0;
}
/*O/P:
4 * 1 = 4
4 * 2 = 8
4 * 3 = 12
...
4 * 20 = 80
*/
