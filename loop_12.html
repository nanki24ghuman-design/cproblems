#include 

int main() {
    char choice;
    int a, b;
    while(1) {
        printf("\nEnter your choice:\n");
        printf("1.Add  2.Subtract  3.Multiply  4.Divide  5.Modulo  Q.Quit\n");
        scanf(" %c", &choice);
        if(choice == 'Q' || choice == 'q')
            break;
        printf("Enter two numbers: ");
        scanf("%d%d", &a, &b);
        switch(choice) {
            case '1': printf("Result = %d\n", a + b); break;
            case '2': printf("Result = %d\n", a - b); break;
            case '3': printf("Result = %d\n", a * b); break;
            case '4': 
                if(b != 0) printf("Result = %d\n", a / b);
                else printf("Division by zero not allowed\n");
                break;
            case '5': 
                if(b != 0) printf("Result = %d\n", a % b);
                else printf("Modulo by zero not allowed\n");
                break;
            default: printf("Invalid choice!\n");
        }
    }
    printf("Program Ended.\n");
    return 0;
}   /* Enter your choice:
