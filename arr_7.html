//enter an array from the user and square all the values of the array?
#include<stdio.h>
void main()
{
int arr[10],i,n;
printf("Enter size of array:");
scanf("%d",&n);
printf("Enter %d elements: ",n);
for(i=0;i<n;i++){
scanf("%d",&arr[i]);
}
for(i=0;i<n;i++){
arr[i]=arr[i]*arr[i];
}
printf("squares are ");
for(i=0;i<n;i++){
printf(" %d",arr[i]);
}
return 0;
}/*Enter size of array: 4  
Enter 4 elements: 2 3 4 5  
Squared values: 4 9 16 25  */
