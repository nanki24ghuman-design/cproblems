//find minimum values of array?
#include<stdio.h>
void main()
{
int arr[5];
int i,min;
printf("Enter 5 numbers  ");
for(i=0;i<5;i++){
scanf("%d",&arr[i]);
}
min=arr[0];
for(i=1;i<5;i++){
if(arr[i]<min){
min=arr[i];
}}
printf("minimum value of array is %d",min);
return 0;
}/*Enter 5 elements: 8 2 10 4 6  
Minimum value: 2  */
