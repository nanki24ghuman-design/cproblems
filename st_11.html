//11. Enter the string from user and print each character in new line ?
#include <stdio.h>

void  main() {
    char str[100];
    int i;
    printf("Enter a string: ");
    scanf("%s",str);

    printf("\nCharacters in string:\n");

    for(i = 0; str[i] != '\0'; i++)
        printf("%c\n", str[i]);  // print each character on new line

    return 0;
}
