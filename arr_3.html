// enter an array from the user and copy it into second array?
#include<stdio.h>
void main()
{
int arr[10],arr1[10],n,i;
printf("Enter size of array:");
scanf("%d",&n);
printf("Enter %d elements: ",n);
for(i=0;i<n;i++){
scanf("%d",&arr[i]);
}
printf("Elements of the second array are:");
for(i=0;i<n;i++){
arr1[i]=arr[i];
}
for(i=0;i<n;i++){
printf("%d ",arr1[i]);
}
return 0;
}/*Enter size of array: 4  
Enter 4 elements: 5 10 15 20  
Copied array: 5 10 15 20  */
